cmake_minimum_required(VERSION 3.21)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
project(NexusUltimate VERSION 0.2.0)
add_library(${PROJECT_NAME} SHARED src/main.cpp src/config.cpp src/rendering.cpp src/gpu_cull.cpp src/frame_gate.cpp src/gpu_timer.cpp src/system_tuning.cpp src/ui.cpp)
if (NOT DEFINED ENV{GEODE_SDK})
 message(FATAL_ERROR "Unable to find Geode SDK! Please define GEODE_SDK environment variable")
endif()
add_subdirectory($ENV{GEODE_SDK} ${CMAKE_BINARY_DIR}/geode-sdk)
setup_geode_mod(${PROJECT_NAME})
target_include_directories(${PROJECT_NAME} PRIVATE src)
if (WIN32)
 target_link_libraries(${PROJECT_NAME} winmm opengl32)
endif()
